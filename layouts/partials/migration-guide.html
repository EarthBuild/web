Adopt EarthBuild in Under 5 Minutes

Adopt EarthBuild alongside your existing toolchain without disrupting active pipelines or rewriting your repository. Start with a single target and migrate at your own pace.

Standard Dockerfile 
FROM golang:1.27-alpine3.24 AS deps
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download

FROM deps AS test
COPY *.go ./
RUN go test -v ./...

FROM deps AS builder
COPY *.go ./
RUN CGO_ENABLED=0 go build -o bin/app .

FROM alpine:3.24 AS runtime
COPY --from=builder /app/bin/app /usr/local/bin/app
ENTRYPOINT ["/usr/local/bin/app"]
Earthfile (Direct Host Artifacts & Push Built-In) 
VERSION 0.8
FROM golang:1.27-alpine3.24
WORKDIR /app

deps:
  COPY go.mod go.sum ./
  RUN go mod download

test:
  FROM +deps
  COPY *.go ./
  RUN go test -v ./...

build:
  FROM +deps
  COPY *.go ./
  RUN CGO_ENABLED=0 go build -o bin/app .
  SAVE ARTIFACT bin/app AS LOCAL bin/app

docker:
  FROM alpine:3.24
  COPY +build/bin/app /usr/local/bin/app
  ENTRYPOINT ["/usr/local/bin/app"]
  SAVE IMAGE --push docker.io/myorg/myapp:latest

all:
  BUILD +test
  BUILD +build
  BUILD +docker
