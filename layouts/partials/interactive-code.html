VERSION 0.8
FROM python:3.14-slim
WORKDIR /app

# deps caches pip dependencies in isolated container layer
deps:
  COPY requirements.txt .
  RUN --mount=type=cache,target=/root/.cache/pip pip install -r requirements.txt

# test runs unit tests inside isolated container
test:
  FROM +deps
  COPY src src
  RUN pytest src/
